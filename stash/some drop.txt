    from pathlib import Path

def get_keyboard(self) -> Keyboard:
    # 1. Read files and extract named instances:
    # Structure: { instance_name: {"class_name": cls_name, "data": dict_data} }
    instances_data = {}
    
    for path in self.expand_paths(CONFIG_DIR):
        class_name = Path(path).stem  # File name = Class name (e.g., 'prompt')
        file_dicts = self.data.get_as_dicts(path)  # Map of instance_name -> instance_dict
        
        for instance_name, data in file_dicts.items():
            instances_data[instance_name] = {
                "class_name": class_name,
                "data": data,
                # Extract all named dependencies mentioned across any field in this instance
                "prereqs": {
                    req_name 
                    for val in data.values() 
                    if isinstance(val, list) 
                    for req_name in val
                }
            }

    done = {}  # Holds instantiated objects: { instance_name: instance_object }
    pending = dict(instances_data)

    # 2. Dependency Resolution Loop
    while pending:
        resolved_this_round = []

        for name, info in pending.items():
            # Check if all prerequisite instance names exist in `done`
            if info["prereqs"].issubset(done.keys()):
                resolved_this_round.append(name)

        # If no new instances could be resolved, we have a circular or missing dependency
        if not resolved_this_round:
            missing = {name: info["prereqs"] - done.keys() for name, info in pending.items()}
            raise RuntimeError(f"Unresolved dependencies detected: {missing}")

        # Bulk instantiation pass for ready items
        newly_instantiated = []
        for name in resolved_this_round:
            info = pending.pop(name)
            
            # Lookup the class by filename/class_name
            cls = _get_class(info["class_name"])
            
            # Pass full `done` context map to the instance (or child of BaseClass)
            instance = cls(name=name, data=info["data"], context=done)
            
            done[name] = instance
            newly_instantiated.append(instance)

        # 3. Bulk operations on newly instantiated state
        for instance in newly_instantiated:
            if hasattr(instance, "on_post_instantiate"):
                instance.on_post_instantiate()

    return Keyboard(done)
